// DOM Cache const monthInput = document.getElementById("monthInput"); const tableBody = document.getElementById("sheetTableBody"); const totalRecordsEl = document.getElementById("totalRecords"); const pickupRecordsEl = document.getElementById("pickupRecords"); const returnRecordsEl = document.getElementById("returnRecords"); // Load Data async function loadSheetData() { const month = monthInput.value.trim(); if (!month) { alert( "Enter Month (Example: 06/2026)" ); monthInput.focus(); return; } // Validation MM/YYYY const monthRegex = /^(0[1-9]|1[0-2])\/\d{4}$/; if (!monthRegex.test(month)) { alert( "Month format should be MM/YYYY" ); return; } showLoading(); try { const result = await getSheetData(month); if (!result.success) { showMessage( result.message || "Failed To Load Data", "danger" ); return; } renderData( result.data || [] ); } catch (error) { console.error(error); showMessage( "Server Error", "danger" ); } } // Loading State function showLoading() { tableBody.innerHTML = `
Loading Data... `; } // Message State function showMessage( message, color = "secondary" ) { tableBody.innerHTML = ` ${message} `; } // Render Table function renderData(data) { let pickup = 0; let ret = 0; if (!data.length) { showMessage( "No Records Found" ); updateSummary( 0, 0, 0 ); return; } const rows = data.map(row => { if (row[3] === "Pickup") pickup++; if (row[3] === "Return") ret++; return ` ${row[0] || "-"} ${row[1] || "-"} ${row[2] || "-"} ${row[3]} ${row[4] || "-"} `; }).join(""); tableBody.innerHTML = rows; updateSummary( data.length, pickup, ret ); } // Update Summary function updateSummary( total, pickup, ret ) { totalRecordsEl.textContent = total; pickupRecordsEl.textContent = pickup; returnRecordsEl.textContent = ret; } // Enter Key Support monthInput.addEventListener( "keydown", function (e) { if (e.key === "Enter") { loadSheetData(); } } );
