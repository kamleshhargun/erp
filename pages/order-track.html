// DOM Elements const trackInput = document.getElementById("trackAwb"); const timelineDiv = document.getElementById("timeline"); const showAwb = document.getElementById("showAwb"); const showOrder = document.getElementById("showOrder"); const showCourier = document.getElementById("showCourier"); const showStatus = document.getElementById("showStatus"); // Search Button Function async function searchTracking() { const awb = trackInput.value.trim(); if (!awb) { alert("Please Enter AWB Number"); trackInput.focus(); return; } await loadTracking(awb); } // Main Tracking Function async function loadTracking(awb) { try { timelineDiv.innerHTML = `
Fetching Tracking Details...
`; const data = await window.trackAWB(awb); if (!data?.success) { timelineDiv.innerHTML = `
${data?.message || "Tracking Not Found"}
`; clearTrackingDetails(); return; } const shipment = data.shipment || {}; renderTrackingDetails( awb, shipment ); renderTimeline( shipment.tracking_events || [] ); } catch (error) { console.error(error); timelineDiv.innerHTML = `
Failed To Fetch Tracking
`; } } // Tracking Details function renderTrackingDetails( awb, shipment ) { showAwb.textContent = awb; showOrder.textContent = shipment.order_id || shipment.orderId || shipment.reference_number || shipment.reference_no || "-"; showCourier.textContent = shipment.courier || "BOXD"; showStatus.innerHTML = getStatusBadge( shipment.status || "-" ); } // Status Badge function getStatusBadge(status) { const value = status.toLowerCase(); let color = "secondary"; if (value.includes("deliver")) color = "success"; else if (value.includes("ofd")) color = "warning"; else if (value.includes("ndr")) color = "danger"; else if (value.includes("rto")) color = "dark"; else if (value.includes("transit")) color = "primary"; return ` ${status} `; } // Timeline function renderTimeline(events) { if (!events.length) { timelineDiv.innerHTML = `
No Timeline Available
`; return; } timelineDiv.innerHTML = events.map(event => `
${event.status || "-"}
${event.created_at || ""} ${ event.location ? `
${event.location}
` : "" }
`).join(""); } // Clear Details function clearTrackingDetails() { showAwb.textContent = "-"; showOrder.textContent = "-"; showCourier.textContent = "-"; showStatus.textContent = "-"; } // Enter Key Support trackInput.addEventListener( "keydown", async (e) => { if (e.key === "Enter") { await searchTracking(); } } );
